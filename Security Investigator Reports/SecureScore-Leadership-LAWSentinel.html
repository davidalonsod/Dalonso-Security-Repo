KQL Queries Behind This Report

1 · 90-day Score Trend

SecureScores
| where TimeGenerated > ago(90d)
| where DisplayName == "ASC score"
| where IsSnapshot == false
| project TimeGenerated, CurrentScore, MaxScore,
          PercentageScore = round(PercentageScore * 100, 1)
| order by TimeGenerated asc

2 · Current Control Scorecard

SecureScoreControls
| where IsSnapshot == false
| summarize arg_max(TimeGenerated, *) by ControlName
| where Weight > 0
| extend Status = case(
    PercentageScore == 1.0, "Complete",
    PercentageScore >= 0.7, "Strong",
    PercentageScore >= 0.4, "Partial",
    PercentageScore > 0,   "Weak",
                           "Not Started")
| project ControlName, Status, CurrentScore, MaxScore,
          HealthyResources, UnhealthyResources, Weight
| order by PercentageScore asc

3 · Unhealthy Backlog (7d)

SecurityRecommendation
| where TimeGenerated > ago(7d)
| where RecommendationState == "Unhealthy"
| summarize AffectedResources = dcount(AssessedResourceId),
            LatestSeen = max(TimeGenerated)
          by RecommendationDisplayName, RecommendationSeverity
| extend SevOrder = case(RecommendationSeverity == "High", 1,
                         RecommendationSeverity == "Medium", 2,
                         RecommendationSeverity == "Low", 3, 4)
| order by SevOrder asc, AffectedResources desc

4 · Remediated Items (Unhealthy → Healthy, 14d)

SecurityRecommendation
| where TimeGenerated > ago(14d)
| summarize States = make_set(RecommendationState),
            First = min(TimeGenerated), Last = max(TimeGenerated)
          by AssessedResourceId, RecommendationDisplayName, RecommendationSeverity
| where States has "Unhealthy" and States has "Healthy"
| summarize Remediated = count() by RecommendationSeverity
| order by Remediated desc
